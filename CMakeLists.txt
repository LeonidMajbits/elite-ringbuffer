cmake_minimum_required(VERSION 3.21)
project(elite_ringbuffer VERSION 1.1.0 LANGUAGES C)
include(GNUInstallDirs)

find_package(Threads REQUIRED)
option(ELITE_BUILD_STATIC "Build static library" ON)
option(ELITE_BUILD_SHARED "Build shared library" ON)
option(ELITE_BUILD_DEMO "Build live POSIX request/reply demo" ON)
option(ELITE_BUILD_TESTS "Build scoped native CTest suite" OFF)
option(ELITE_EXPERIMENTAL_FREEBSD "Admit experimental FreeBSD POLL_ONLY profile" OFF)
option(ELITE_STRICT_LDAR "Disable RCpc selection on Apple ARM64" ON)
if(NOT ELITE_BUILD_STATIC AND NOT ELITE_BUILD_SHARED)
  message(FATAL_ERROR "At least one library linkage is required")
endif()
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Darwin|FreeBSD)$")

  message(FATAL_ERROR "FreeBSD is experimental: pass -DELITE_EXPERIMENTAL_FREEBSD=ON")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64|aarch64|arm64)$")
  message(FATAL_ERROR "Only little-endian 64-bit x86/ARM profiles are admitted")
endif()
if(APPLE)

    message(FATAL_ERROR "The public Darwin address-wait adapter requires macOS >= 14.4")
  endif()

set(ELITE_SOURCES src/elite_core.c src/elite_format.c src/elite_spsc.c
  src/elite_mpmc_ncq.c src/elite_shm.c src/elite_wait.c src/elite_version.c)
function(elite_compile target)
  set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO POSITION_INDEPENDENT_CODE ON)
  target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic
    -Wconversion -Wsign-conversion -Wshadow -Wstrict-prototypes -Wmissing-prototypes -Wformat=2 -Wundef)
  if(APPLE)
    target_compile_definitions(${target} PRIVATE _DARWIN_C_SOURCE)
    if(ELITE_STRICT_LDAR AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
      target_compile_options(${target} PRIVATE -Xclang -target-feature -Xclang -rcpc)
    endif()
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(${target} PRIVATE _DEFAULT_SOURCE)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
    target_compile_definitions(${target} PRIVATE ELITE_EXPERIMENTAL_FREEBSD=1)
  endif()
endfunction()
set(ELITE_INSTALL_TARGETS)
foreach(kind IN ITEMS STATIC SHARED)
  if(ELITE_BUILD_${kind})
    string(TOLOWER ${kind} lower)
    add_library(elite_ringbuffer_${lower} ${kind} ${ELITE_SOURCES})
    elite_compile(elite_ringbuffer_${lower})
    target_include_directories(elite_ringbuffer_${lower}
      PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
      PRIVATE src)
    target_link_libraries(elite_ringbuffer_${lower} PUBLIC Threads::Threads)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      target_link_libraries(elite_ringbuffer_${lower} PRIVATE rt)
    endif()
    set_target_properties(elite_ringbuffer_${lower} PROPERTIES OUTPUT_NAME elite_ringbuffer EXPORT_NAME ringbuffer_${lower})
    if(kind STREQUAL "SHARED")
      set_target_properties(elite_ringbuffer_shared PROPERTIES VERSION 1.1.0 SOVERSION 1)
      if(APPLE)
        set_target_properties(elite_ringbuffer_shared PROPERTIES INSTALL_NAME_DIR "@rpath" MACOSX_RPATH ON)
      endif()
    endif()
    add_library(elite::ringbuffer_${lower} ALIAS elite_ringbuffer_${lower})
    list(APPEND ELITE_INSTALL_TARGETS elite_ringbuffer_${lower})
  endif()
endforeach()
add_library(elite_ringbuffer INTERFACE)
if(ELITE_BUILD_SHARED)
  target_link_libraries(elite_ringbuffer INTERFACE elite_ringbuffer_shared)
else()
  target_link_libraries(elite_ringbuffer INTERFACE elite_ringbuffer_static)
endif()

add_library(elite::ringbuffer ALIAS elite_ringbuffer)
list(APPEND ELITE_INSTALL_TARGETS elite_ringbuffer)
