cmake_minimum_required(VERSION 3.21)
project(elite_external_consumer LANGUAGES C CXX)
find_package(elite_ringbuffer 1.1 CONFIG REQUIRED)
add_executable(consumer consumer.c)
target_link_libraries(consumer PRIVATE elite::ringbuffer)
add_executable(consumer_cpp consumer.cpp)
target_compile_features(consumer_cpp PRIVATE cxx_std_20)
target_link_libraries(consumer_cpp PRIVATE elite::ringbuffer)
enable_testing()
add_test(NAME installed_c COMMAND consumer)
add_test(NAME installed_cpp COMMAND consumer_cpp)
